\title[The Gaetz--Gao closure conjecture]{The Gaetz--Gao Closure Conjecture for Finite Coxeter Systems}
\date{}

\subjclass[2020]{Primary 20F55; Secondary 05E14, 20-08}
\keywords{Billey--Postnikov decomposition, finite Coxeter system,
type $H_4$, exact enumeration, computer-assisted proof}

\begin{document}

\begin{abstract}
Gaetz and Gao conjectured that, for every Coxeter system $(W,S)$ and
$w\in W$, the family of parabolic subsets giving Billey--Postnikov
decompositions of $w$ is closed under union and intersection. They proved
the conjecture for finite Weyl groups and for Coxeter systems of rank
three. We prove it for every finite Coxeter system. A direct-product
reduction, an elementary argument in rank at most two, and the
classification of finite Coxeter systems leave only type
$H_4$. An exact exhaustive computation over its $14{,}400$ elements checks
all $230{,}400$ parabolic decompositions and, summed over those elements, all
$201{,}991$ unordered pairs of distinct Billey--Postnikov subsets, with no
closure failure.
\end{abstract}

\maketitle

\section{The finite reduction}

Let $(W,S)$ be a Coxeter system, with length function $\ell$. For
$x\in W$, write
\[
  \DL(x)=\{s\in S:\ell(sx)<\ell(x)\},
  \qquad
  \DR(x)=\{s\in S:\ell(xs)<\ell(x)\},
\]
and let $\Supp(x)\subseteq S$ be the support of $x$, that is, the set of
generators occurring in some (equivalently, in every) reduced word for $x$.
For $J\subseteq S$ let $W_J=\langle J\rangle$ be the corresponding standard
parabolic subgroup and let
\[
  W^J=\{x\in W:\ell(xs)>\ell(x)\text{ for all }s\in J\}
     =\{x\in W:\DR(x)\cap J=\varnothing\}
\]
be the set of minimal length representatives of the cosets $xW_J$. The
parabolic decomposition of $w$ with respect to $J$ is the unique
factorization
\[
  w=w^Jw_J,
  \qquad
  w^J\in W^J,
  \quad
  w_J\in W_J,
  \quad
  \ell(w)=\ell(w^J)+\ell(w_J).
\]
Following Gaetz and Gao \cite[Definition~2.1]{GaetzGao}, set
\[
  \BP_W(w)
  =
  \bigl\{
    J\subseteq S:
    \Supp(w^J)\cap J\subseteq \DL(w_J)
  \bigr\}.
\]
Gaetz and Gao conjectured that $\BP_W(w)$ is closed under union and
intersection for every Coxeter system \cite[Conjecture~1.3]{GaetzGao}.
They proved this for finite Weyl groups \cite[Theorem~1.2]{GaetzGao} and for
Coxeter systems of rank three \cite[Theorem~3.15]{GaetzGao}, a result they
state as covering rank at most three \cite[Section~1]{GaetzGao}.

\begin{theorem}\label{thm:main}
Let $(W,S)$ be a finite Coxeter system and let $w\in W$. If
$J,K\in\BP_W(w)$, then
\[
  J\cap K\in\BP_W(w)
  \qquad\text{and}\qquad
  J\cup K\in\BP_W(w).
\]
Thus, under inclusion, $\BP_W(w)$ is a distributive sublattice of $2^S$.
\end{theorem}

\begin{lemma}\label{lem:product}
Suppose
\[
  (W,S)=\prod_{i=1}^r (W_i,S_i),
  \qquad
  S=S_1\sqcup\cdots\sqcup S_r,
  \qquad
  w=(w_1,\ldots,w_r).
\]
For $J\subseteq S$, put $J_i=J\cap S_i$. Under the identification
$2^S\cong\prod_i2^{S_i}$,
\[
  \BP_W(w)=\prod_{i=1}^r \BP_{W_i}(w_i).
\]
\end{lemma}

\begin{proof}
Parabolic decomposition, support, and left descents factor componentwise:
\[
  w^J=(w_i^{J_i})_i,
  \qquad
  w_J=((w_i)_{J_i})_i,
\]
\[
  \Supp(w^J)\cap J
  =\bigsqcup_i\bigl(\Supp(w_i^{J_i})\cap J_i\bigr),
  \qquad
  \DL(w_J)=\bigsqcup_i\DL((w_i)_{J_i}).
\]
Hence the Billey--Postnikov condition holds exactly when it holds in every
component.
\end{proof}

The rank-two case needs no computation and no hypothesis of finiteness.

\begin{lemma}\label{lem:rank2}
Let $(W,S)$ be any Coxeter system and let $w\in W$. Then
$\varnothing\in\BP_W(w)$ and $S\in\BP_W(w)$. Consequently, if $|S|\leq2$,
then $\BP_W(w)$ is closed under union and intersection.
\end{lemma}

\begin{proof}
Since $\DR(x)\cap\varnothing=\varnothing$ for every $x$, we have
$W^{\varnothing}=W$, so $w^{\varnothing}=w$ and $w_{\varnothing}=e$, and
therefore
\[
  \Supp(w^{\varnothing})\cap\varnothing
  =\varnothing
  \subseteq\DL(e)=\varnothing .
\]
Since $\DR(x)=\varnothing$ only for $x=e$, we have $W^S=\{e\}$, so $w^S=e$
and $w_S=w$, and therefore
\[
  \Supp(w^S)\cap S=\varnothing\subseteq\DL(w).
\]
Now let $|S|\leq2$ and let $J,K\in\BP_W(w)$. If $J$ and $K$ are nested,
then $J\cup K$ and $J\cap K$ are $J$ and $K$ in some order, hence lie in
$\BP_W(w)$. Otherwise $|S|=2$, say $S=\{s,t\}$, and $\{J,K\}$ must be the
one non-nested pair of subsets of $S$, namely $\{s\}$ and $\{t\}$; then
$J\cup K=S$ and $J\cap K=\varnothing$, which lie in $\BP_W(w)$ by the
previous paragraph.
\end{proof}

By the classification of irreducible finite Coxeter systems
\cite{Humphreys}, the irreducible finite types not already finite Weyl
types are $H_3$, $H_4$, and $I_2(m)$ with $m=5$ or $m\geq7$. Among these,
$H_3$ has rank three and is covered by \cite[Theorem~3.15]{GaetzGao}, while
the dihedral types have rank two and are covered by
Lemma~\ref{lem:rank2}. Together with Lemma~\ref{lem:product} and the
Gaetz--Gao theorem for finite Weyl groups \cite[Theorem~1.2]{GaetzGao},
this reduces Theorem~\ref{thm:main} to $H_4$.

\section{Exact verification in type \texorpdfstring{$H_4$}{H4}}
\label{sec:h4}

We first record the parabolic-decomposition algorithm used by the
verification.

\begin{lemma}\label{lem:stripping}
Fix $w\in W$ and $J\subseteq S$. Set $x_0=w$ and $p_0=e$. While
$\DR(x_t)\cap J\neq\varnothing$, choose $s_t\in\DR(x_t)\cap J$ and set
\[
  x_{t+1}=x_ts_t,
  \qquad
  p_{t+1}=s_tp_t.
\]
If the process stops after $k$ steps, then
\[
  x_k=w^J,
  \qquad
  p_k=w_J.
\]
In particular, the output is independent of the choices of the $s_t$.
\end{lemma}

\begin{proof}
The process terminates because $\ell(x_{t+1})=\ell(x_t)-1$. Its invariant
is $w=x_tp_t$. At termination, $\DR(x_k)\cap J=\varnothing$, so
$x_k\in W^J$, while $p_k\in W_J$. Moreover,
\[
  \ell(x_k)=\ell(w)-k,
  \qquad
  \ell(p_k)\leq k.
\]
Therefore
\[
  \ell(w)=\ell(x_kp_k)
  \leq \ell(x_k)+\ell(p_k)
  \leq \ell(w)-k+k
  =\ell(w).
\]
All inequalities are equalities, and uniqueness of parabolic decomposition
gives the claim.
\end{proof}

Let $W$ be of type $H_4$, with simple reflections
$S=\{s_1,s_2,s_3,s_4\}$ and Coxeter matrix determined by
\[
  m_{12}=5,
  \qquad
  m_{23}=m_{34}=3,
  \qquad
  m_{ij}=2\ \text{otherwise}.
\]
Put $\varphi=(1+\sqrt5)/2$. We use the faithful geometric representation
\cite{Humphreys} on the basis
$\alpha_1,\ldots,\alpha_4$, where
\[
  s_i(\alpha_i)=-\alpha_i,
  \qquad
  s_i(\alpha_j)
  =\alpha_j+2\cos(\pi/m_{ij})\alpha_i
  \quad(i\neq j).
\]
Every matrix entry belongs to $\mathbb Z[\varphi]$. The verifier represents
$a+b\varphi$ by the integer pair $(a,b)$ and uses only
$\varphi^2=\varphi+1$.

The verifier first checks the Coxeter relations and enumerates the generated
matrix group by breadth-first search. It obtains $14{,}400$ matrices. The
breadth-first distances give Coxeter length, and their distribution is
checked against
\[
  [2]_q[12]_q[20]_q[30]_q,
  \qquad
  [d]_q=1+q+\cdots+q^{d-1},
\]
the Poincar\'e polynomial $\sum_{w\in W}q^{\ell(w)}$ of $H_4$: for a finite
Coxeter group this polynomial factors as $\prod_i[d_i]_q$ over the degrees
$d_i$ of the group, and the degrees of $H_4$ are $2,12,20,30$; see
\cite{Humphreys} for both facts. This is the check that identifies the
enumerated matrix group as $W(H_4)$ with breadth-first distance equal to
Coxeter length. The resulting reduced words and exact
left and right multiplication tables then give support and descent sets.
For every $w\in W$ and every $J\subseteq S$,
Lemma~\ref{lem:stripping} computes $w^J$ and $w_J$; the verifier asserts
exact product equality, membership in $W^J$ and $W_J$, and length
additivity. It then tests
\[
  \Supp(w^J)\cap J\subseteq\DL(w_J)
\]
and checks union and intersection for every unordered pair of distinct
members of every resulting family $\BP_W(w)$.

\begin{proposition}\label{prop:h4}
For every $w$ in the Coxeter group of type $H_4$, the family $\BP_W(w)$ is
closed under union and intersection.
\end{proposition}

\begin{proof}[Computer-assisted proof]
The exact computation returns
\[
\begin{array}{lr}
\toprule
\text{elements enumerated} & 14{,}400\\
\text{maximum Coxeter length} & 60\\
\text{BP membership tests} & 230{,}400\\
\text{positive BP memberships} & 78{,}498\\
\text{unordered distinct BP pairs} & 201{,}991\\
\text{union/intersection failures} & 0\\
\bottomrule
\end{array}
\]
and the following distribution:
\[
\begin{array}{c|rrrrrrrrrr}
|\BP_W(w)|
  &3&4&5&6&7&8&9&10&12&16\\ \hline
\#w
  &2492&62&8872&150&65&24&2627&68&24&16.
\end{array}
\]
The second row sums to $14{,}400$; weighting it by $|\BP_W(w)|$ and by
$\binom{|\BP_W(w)|}{2}$ gives $78{,}498$ and $201{,}991$, respectively.
Equal pairs are automatic, so the exhaustive scan proves the proposition.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Gaetz and Gao cover every finite Weyl component
\cite[Theorem~1.2]{GaetzGao} and every component of rank three
\cite[Theorem~3.15]{GaetzGao}; Lemma~\ref{lem:rank2} covers every component
of rank at most two; Proposition~\ref{prop:h4} covers the sole remaining
irreducible finite type; and Lemma~\ref{lem:product} combines the
irreducible components.
\end{proof}

\medskip
\noindent\textbf{Reproducibility.}
Section~\ref{sec:h4} specifies the computation completely: the four
reflection matrices over $\mathbb Z[\varphi]$, the breadth-first
enumeration, and the descent-stripping algorithm of
Lemma~\ref{lem:stripping}. Taking only the Coxeter matrix of $H_4$ from
this paper, a referee can recompute from them every quantity tabulated in
Proposition~\ref{prop:h4}, the $|\BP_W(w)|$ profile and its three
consistency identities, and the length distribution against
$[2]_q[12]_q[20]_q[30]_q$, in exact $\mathbb Z[\varphi]$ arithmetic with no
floating-point operations. The census was carried out twice, by two
independently written programs; the second was re-run on a separate
machine, where all $53$ of its checks agreed, and it also verifies
Theorem~\ref{thm:main} outright for $25$ further finite Coxeter systems,
among them $I_2(5)\times H_4$ ($144{,}000$ elements) and $A_2\times H_4$
($86{,}400$). That re-run does not re-prove the Gaetz--Gao theorem for
finite Weyl groups cited here, and does not reach types $B$, $F$ and
$G_2=I_2(6)$, whose Coxeter matrices need $2\cos(\pi/4)$ or $2\cos(\pi/6)$
and so leave $\mathbb Z[\varphi]$, nor $E_7$ and $E_8$; what it establishes
is agreement with the numbers reported above, not a second proof. The programs and their transcripts are retained in an auxiliary
archive, available on request.

\medskip
\noindent\textbf{Status and scope.}
Theorem~\ref{thm:main} is proved modulo
\cite[Theorem~1.2]{GaetzGao} and \cite[Theorem~3.15]{GaetzGao}, which at the
time of writing are available only in an unrefereed preprint; the
contributions of this note are Lemmas~\ref{lem:product} and~\ref{lem:rank2}
and the exhaustive $H_4$ census of Proposition~\ref{prop:h4}. The printed
hypothesis of \cite[Theorem~3.15]{GaetzGao} is rank three, but the rank-two
dihedral components $I_2(m)$ do not depend on it, nor on the
rank-at-most-three form of that result stated in
\cite[Section~1]{GaetzGao}: Lemma~\ref{lem:rank2} settles them from the
definition alone, for every $m$, and likewise for the infinite dihedral
system. The general Gaetz--Gao conjecture remains open for infinite Coxeter
systems of rank at least four.
